```python
# 数字签名流程（模拟）

import hashlib

class DigitalSignature:
    def __init__(self, private_key):
        self.private_key = private_key

    def sign(self, message):
        """用私钥对消息签名"""
        combined = message + self.private_key
        hash_val = hashlib.sha256(combined.encode()).hexdigest()[:16]
        return hash_val

    def verify(message, signature, public_key):
        """用公钥验证签名"""
        expected = hashlib.sha256((message + public_key).encode()).hexdigest()[:16]
        return signature == expected

# Alice 创建密钥对
private_key = "alice_private_key_xyz"
public_key = "alice_public_key_abc"

# Alice 签名交易
alice = DigitalSignature(private_key)
transaction = "Alice -> Bob: 10 BTC"
signature = alice.sign(transaction)

print(f"交易: {transaction}")
print(f"签名: {signature}")
print(f"\nBob 验证签名: {DigitalSignature.verify(transaction, signature, public_key)}")
```

```python
# Merkle 树构建演示

import hashlib

def hash_pair(left, right):
    """对两个哈希值取配对哈希"""
    combined = left + right
    return hashlib.sha256(combined.encode()).hexdigest()[:8]

# 交易数据
transactions = [
    "Alice -> Bob: 10 BTC",
    "Bob -> Carol: 5 BTC",
    "Carol -> Dave: 3 BTC",
    "Dave -> Eve: 1 BTC",
]

# 第一层：交易哈希
level1 = [hashlib.sha256(t.encode()).hexdigest()[:8] for t in transactions]
print("第1层（交易哈希）:")
for i, h in enumerate(level1):
    print(f"  TX{i+1}: {h}")

# 第二层：配对哈希
level2 = [hash_pair(level1[0], level1[1]), hash_pair(level1[2], level1[3])]
print("\n第2层（配对哈希）:")
for i, h in enumerate(level2):
    print(f"  N{i+1}: {h}")

# 第三层：Merkle 根
merkle_root = hash_pair(level2[0], level2[1])
print(f"\nMerkle 根: {merkle_root}")
print(f"\n作用: 用 Merkle 根即可验证任意交易是否在区块中")
```

```python
import random

class General:
    def __init__(self, name, is_loyal):
        self.name = name
        self.is_loyal = is_loyal
        self.vote = None

        self.vote = "进攻" if attack else "撤退"
```

```python
# 工作量证明 (PoW) 模拟

import hashlib, time

class ProofOfWork:
    def __init__(self, block_data, difficulty=4):
        self.block_data = block_data
        self.difficulty = difficulty  # 前导零个数

    def mine(self):
        """挖矿：寻找满足条件的 nonce"""
        prefix = "0" * self.difficulty
        nonce = 0
        start = time.time()

        while True:
            text = f"{self.block_data}{nonce}"
            hash_result = hashlib.sha256(text.encode()).hexdigest()
            if hash_result.startswith(prefix):
                elapsed = time.time() - start
                return nonce, hash_result, elapsed
            nonce += 1

# 模拟挖矿
block = ProofOfWork("Block #1: Alice->Bob: 10 BTC", difficulty=3)
nonce, hash_result, elapsed = block.mine()

print(f"挖矿成功!")
print(f"Nonce: {nonce}")
print(f"哈希: {hash_result}")
print(f"耗时: {elapsed:.4f} 秒")
print(f"\n难度: {block.difficulty} 个前导零")
print(f"特点: 难度越高 → nonce 搜索空间越大 → 耗时长")
```

```python
# AMM (Automated Market Maker) 原理

print("=== 恒定乘积公式 x * y = k ===\n")

class SimpleAMM:
    def __init__(self, x_reserve, y_reserve):
        self.x = x_reserve
        self.y = y_reserve
        self.k = x_reserve * y_reserve  # 恒定乘积

    def get_price(self, x_in):
        """计算用 x 换 y 的价格"""
        new_x = self.x + x_in * 0.97  # 0.3% 手续费
        new_y = self.k / new_x
        y_out = self.y - new_y
        return y_out

    def add_liquidity(self, x, y):
        self.x += x
        self.y += y
        self.k = self.x * self.y

# 初始化: ETH/USDC 池
amm = SimpleAMM(x_reserve=100, y_reserve=200000)

print(f"初始: ETH={amm.x}, USDC={amm.y}, k={amm.k:.0f}")

# 用 1 ETH 换 USDC
usdc_out = amm.get_price(1)
print(f"\n用 1 ETH 换得: {usdc_out:.2f} USDC")
print(f"兑换后: ETH={amm.x + 1 * 0.97:.2f}, USDC={amm.y - usdc_out:.2f}")

# 添加流动性
amm.add_liquidity(10, 20000)
print(f"\n添加流动性后: ETH={amm.x}, USDC={amm.y}, k={amm.k:.0f}")
```

```python
import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/token/ERC721/extensions/ERC721URIStorage.sol";

contract MyNFT is ERC721, ERC721URIStorage {
    uint256 private _tokenIdCounter;
    uint256 public maxSupply;
    uint256 public mintPrice;

    constructor()
        ERC721("MyNFT", "MNFT")
    {
        maxSupply = 10000;
        mintPrice = 0.05 ether;
        _tokenIdCounter = 0;
    }

    function mint() external payable {
        require(_tokenIdCounter < maxSupply, "Sold out");
        require(msg.value >= mintPrice, "Insufficient funds");

        uint256 tokenId = _tokenIdCounter++;
        _safeMint(msg.sender, tokenId);
        _setTokenURI(tokenId, generateTokenURI(tokenId));
    }

    function generateTokenURI(uint256 tokenId)
        internal
        pure
        returns (string memory)
    {
        // 返回 JSON 元数据的 IPFS 链接
        return string(abi.encodePacked(
            "ipfs://Qm...", Strings.toString(tokenId), ".json"
        ));
    }
}
"""
```
