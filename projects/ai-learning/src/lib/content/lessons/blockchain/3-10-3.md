# AMM 与流动性池

## 自动做市商 (AMM)

AMM（Automated Market Maker）是 DeFi 的核心创新，用数学公式替代传统订单簿，任何人都可以成为做市商。

```python
# 恒定乘积 AMM 模拟
print("=== 恒定乘积公式 x * y = k ===\n")

class SimpleAMM:
    def __init__(self, x_reserve, y_reserve, fee=0.003):
        self.x = x_reserve
        self.y = y_reserve
        self.k = x_reserve * y_reserve
        self.fee = fee
    
    def get_price(self, x_in):
        """用 x 兑换 y，返回得到的 y 数量"""
        x_with_fee = x_in * (1 - self.fee)
        new_x = self.x + x_with_fee
        new_y = self.k / new_x
        return self.y - new_y
    
    def add_liquidity(self, x, y):
        """添加流动性"""
        self.x += x
        self.y += y
        self.k = self.x * self.y
    
    def get_lp_tokens(self, x, y):
        """计算获得的 LP 代币数量"""
        return (x * y) ** 0.5

# 初始化 ETH/USDC 池
amm = SimpleAMM(x_reserve=100, y_reserve=200000)

print(f"初始: ETH={amm.x}, USDC={amm.y}, k={amm.k:.0f}")
print(f"初始价格: 1 ETH = {amm.y/amm.x:.0f} USDC\n")

# 用 1 ETH 换 USDC
usdc_out = amm.get_price(1)
print(f"用 1 ETH 兑换: {usdc_out:.2f} USDC")
print(f"兑换后: ETH={amm.x + 0.997:.2f}, USDC={amm.y - usdc_out:.2f}")
print(f"新价格: 1 ETH = {(amm.y - usdc_out)/(amm.x + 0.997):.0f} USDC")
print("\n注意: 兑换后价格变了！大额兑换滑点大。")
```

## 无常损失

```python
# 无常损失计算
print("=== 无常损失 (Impermanent Loss) ===\n")

# 价格变化场景
price_changes = [1.0, 1.5, 2.0, 3.0, 0.5, 0.33]
print("价格变化 → LP 损失 vs 持有:")
print(f"  {'价格变化':>10} {'LP价值':>10} {'持有价值':>10} {'IL损失':>10}")
print(f"  {'':10} {'(50/50)':>10} {'(HODL)':>10} {'':>10}")

for change in price_changes:
    hodl = 100 * (1 + change) / 2  # 持有价值
    lp = 200 / (change ** 0.5)      # LP 价值
    il = (hodl - lp) / hodl * 100
    print(f"  {change:>+10.0%} {lp:>10.1f} {hodl:>10.1f} {il:>+9.1f}%")

print("\nIL 随价格偏离加剧，但手续费收入可部分抵消")
```

## 知识点总结

- AMM 核心：x * y = k（恒定乘积公式）
- 滑点：大额交易导致价格变化
- LP（流动性提供者）：赚取交易手续费
- 无常损失：价格变化导致 LP 价值低于持有
