/**
 * PyodideRunner — Python 代码执行器组件
 *
 * 封装 usePyodide Hook，提供 Python 代码执行能力。
 * 当代码变化或手动触发时，自动执行代码并将结果返回给父组件。
 *
 * @param code - 要执行的 Python 代码
 * @param onRunComplete - 执行完成后的回调，接收 { output, error }
 * @param autoRun - 是否在 code 变化时自动执行（默认 false）
 */
 
import { useEffect, useCallback } from 'react';
import { usePyodide, type PyodideResult } from '@/hooks/usePyodide';
 
/**
 * PyodideRunner 组件的属性
 */
export interface PyodideRunnerProps {
  /** 要执行的 Python 代码 */
  code: string;
  /** 执行完成后的回调函数，接收执行结果 */
  onRunComplete: (result: { output: string; error: string }) => void;
  /** 是否在 code 变化时自动执行（默认 false，需手动触发） */
  autoRun?: boolean;
}
 
/**
 * PyodideRunner 组件
 *
 * 内部使用 usePyodide Hook 管理 Pyodide 生命周期：
 * 1. 组件挂载时自动加载 Pyodide
 * 2. 接收到代码后执行 Python
 * 3. 将 stdout/stderr 通过回调传递给父组件
 *
 * @param props - 组件属性
 * @returns null（无 UI 渲染，纯逻辑组件）
 */
export default function PyodideRunner({
  code,
  onRunComplete,
  autoRun = false,
}: PyodideRunnerProps) {
  // 使用 usePyodide Hook 获取执行能力
  const { isLoading, error: pyodideError, runPython } = usePyodide();
 
  /**
   * 执行代码的内部处理函数
   */
  const executeCode = useCallback(async () => {
    // Pyodide 还在加载中，不执行
    if (isLoading) {
      onRunComplete({
        output: '',
        error: 'Python 运行环境正在加载中，请稍候...',
      });
      return;
    }
 
    // Pyodide 加载出错
    if (pyodideError) {
      onRunComplete({
        output: '',
        error: `Python 运行环境加载失败：${pyodideError}`,
      });
      return;
    }
 
    // 空代码不执行
    if (!code.trim()) {
      onRunComplete({ output: '', error: '' });
      return;
    }
 
    // 执行 Python 代码
    const result: PyodideResult = await runPython(code);
    onRunComplete({
      output: result.stdout,
      error: result.stderr,
    });
  }, [code, isLoading, pyodideError, runPython, onRunComplete]);
 
  /**
   * 当 autoRun 为 true 时，code 变化自动执行
   */
  useEffect(() => {
    if (autoRun) {
      executeCode();
    }
  }, [autoRun, executeCode]);
 
  // 将执行函数暴露给父组件（通过 ref 或状态）
  // 这里使用 useEffect 监听外部触发（通过 code 变化 + autoRun）
 
  return null; // 纯逻辑组件，不渲染任何 UI
}
