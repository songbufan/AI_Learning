/**
 * 根据章节 ID 获取章节数据
 *
 * @param chapterId - 章节 ID
 * @returns 匹配的 Chapter 对象，未找到返回 undefined
 */
export function getChapterById(chapterId: number): Chapter | undefined {
  return chapters.find((ch) => ch.id === chapterId);
}
 
/**
 * 根据课时 ID 获取课时数据
 *
 * @param lessonId - 课时 ID，格式为 "章节号-课时号"，如 "1-1"
 * @returns 匹配的 Lesson 对象，未找到返回 undefined
 */
export function getLessonById(lessonId: string): Lesson | undefined {
  for (const chapter of chapters) {
    const lesson = chapter.lessons.find((l) => l.id === lessonId);
    if (lesson) return lesson;
  }
  return undefined;
}
 
/**
 * 获取所有课时列表（扁平化）
 *
 * @returns 包含所有课时的一维数组
 */
export function getAllLessons(): Lesson[] {
  return chapters.flatMap((ch) => ch.lessons);
}
